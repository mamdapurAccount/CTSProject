@Path("/hello")
public class HelloService {
	static DbUtil db = new DbUtil();
	@GET
	@Path("/msg")
	@Produces(MediaType.TEXT_PLAIN)
	public String sayHello() {
		return "Dada I Miss you";
	}
	@GET
	@Path("/person")
	@Produces(MediaType.APPLICATION_XML)
	public Person getPerson() {
		return new Person("Murali","Krishna",90);
	}
	@POST
	@Path("/addperson")
	@Consumes(MediaType.APPLICATION_JSON)
	@Produces(MediaType.APPLICATION_JSON)
	public String getName(Person p) {
		return p.getFname()+"  "+p.getLname();
	}
	@GET
	@Path("/getusers")
	@Produces(MediaType.APPLICATION_JSON)
	public List<Login> getAllusers() {
		return db.getUsers();
	}
	
	@POST
	@Path("/adduser")
	@Produces(MediaType.APPLICATION_JSON)
	@Consumes(MediaType.TEXT_PLAIN)
	public String AddUser(Login l) {
		return db.addUser(l);
	}
	@GET
	@Path("/getuser/{p1}")
	@Produces(MediaType.APPLICATION_JSON)
	@Consumes(MediaType.APPLICATION_JSON)
	
	public Login getUser(@PathParam("p1") String s) {
		return db.getUser(s);
	}
	@PUT
	@Path("/updateuser")
	@Produces(MediaType.APPLICATION_JSON)
	@Consumes(MediaType.APPLICATION_JSON)
	public String updateUser(Login l) {
		return db.updatePwd(l);
	}
	@DELETE
	@Path("/deluser/{p1}")
	@Produces(MediaType.APPLICATION_JSON)
	@Consumes(MediaType.APPLICATION_JSON)
	public String delUser(@PathParam("p1") String u) {
		return db.delUser(u);
	}
	@GET
	@Path("/getmyusers")
	@Produces(MediaType.APPLICATION_JSON)
	public Response getUsers() {
		return Response.ok("[{\"username\":lolo},{\"username\":bebo}]").build();
	}
	@POST
	@Path("/addmyuser")
	@Consumes(MediaType.APPLICATION_JSON)
	@Produces(MediaType.TEXT_PLAIN)
	public Response addUser(String str) {
          return Response.status(201).entity("User Created "+str).build();
	}
	@GET
	@Path("/getmyuser/{p1}")
	@Consumes(MediaType.APPLICATION_JSON)
	@Produces(MediaType.TEXT_PLAIN)
	public Response getUser(@PathParam("p1") int id) {
		if(id<=5)
          return Response.status(401).entity("Invalid Id").build();
		else
			return Response.status(201).entity("User "+id).build();
	}
	
	@GET
	@Path("/admin")
	@RolesAllowed("ADMIN")
	public Response adminOnly() {
	    return Response.ok("Admin access granted").build();
	}

	@OPTIONS
    @Path("/getmyusers")
	@Produces(MediaType.APPLICATION_JSON)
    public Response handleOptions() {
        return Response.ok()
                .header("Access-Control-Allow-Origin", "*")
                .header("Access-Control-Allow-Methods", "GET, POST, PUT, DELETE, OPTIONS")
                .header("Access-Control-Allow-Headers", "Content-Type, Authorization")
                .build();
    }

	@GET
	@Path("/myadmin")
	public Response adminOnly(@HeaderParam("Authorization") String authHeader) {
		
		return Response.ok(authHeader).build();
